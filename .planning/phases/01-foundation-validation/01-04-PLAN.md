Replace the current render_editor_content() with uniform_list-based rendering:

```rust
fn render_editor_content(&self, cx: &Context<Self>) -> impl IntoElement {
    let theme = current_theme();

    match self.active_tab() {
        Some(tab) if !tab.is_loading => {
            let document = tab.document.clone();
            let line_count = document.read(cx).line_count();

            div()
                .id("editor-content")
                .flex_1()
                .overflow_hidden()  // Required for uniform_list
                .bg(theme.background.editor)
                .p(px(8.0))
                .font_family(theme.fonts.mono_family.clone())
                .text_size(px(theme.fonts.mono_size))
                .child(
                    uniform_list(
                        cx.entity().clone(),
                        "editor-lines",
                        line_count,
                        move |_pane, visible_range, _window, cx| {
                            let doc = document.read(cx);
                            visible_range
                                .map(|line_idx| {
                                    Self::render_line(line_idx, &doc, &theme)
                                })
                                .collect()
                        },
                    )
                    .track_scroll(&self.scroll_handle)
                )
        }
        Some(_tab) => {
            // Loading state
            div()
                .id("loading")
                .flex_1()
                .flex()
                .items_center()
                .justify_center()
                .bg(theme.background.editor)
                .text_color(theme.text.secondary)
                .child("Loading...")
        }
        None => {
            // Empty state
            div()
                .id("empty-editor")
                .flex_1()
                .flex()
                .items_center()
                .justify_center()
                .bg(theme.background.editor)
                .text_color(theme.text.secondary)
                .child("No file open - Press Cmd+O to open a file")
        }
    }
}
```

Add the render_line helper function:

```rust
fn render_line(line_idx: usize, doc: &Document, theme: &Theme) -> impl IntoElement {
    let line_text = doc.line(line_idx).unwrap_or_default();
    // Trim trailing newline for display
    let display_text = line_text.trim_end_matches('\n');

    div()
        .h(px(LINE_HEIGHT))  // CRITICAL: uniform height for all lines
        .w_full()
        .flex()
        .child(
            // Line number gutter
            div()
                .w(px(50.0))
                .h_full()
                .flex()
                .items_center()
                .justify_end()
                .pr(px(12.0))
                .text_color(theme.text.line_number)
                .child(SharedString::from(format!("{}", line_idx + 1)))
        )
        .child(
            // Line content
            div()
                .flex_1()
                .h_full()
                .flex()
                .items_center()
                .text_color(theme.text.primary)
                .child(if display_text.is_empty() {
                    SharedString::from(" ")  // Empty lines need content for height
                } else {
                    SharedString::from(display_text.to_string())
                })
        )
}
```

Key points:
- uniform_list only calls the render callback for visible lines
- LINE_HEIGHT MUST be consistent for all lines (uniform_list requirement)
- .track_scroll() connects scroll handle for position tracking
- visible_range gives the indices of lines to render
